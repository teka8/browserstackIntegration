CREATE OR ALTER PROCEDURE Refresh_Monthly_Orders_Incremental
AS
BEGIN
    SET NOCOUNT ON;

    ---------------------------
    -- 1) Calculate date ranges
    ---------------------------

    -- Month to delete (3 months old)
    DECLARE @DeleteMonthStart DATE = DATEFROMPARTS(
        YEAR(DATEADD(MONTH, -3, GETDATE())),
        MONTH(DATEADD(MONTH, -3, GETDATE())),
        1
    );

    DECLARE @DeleteMonthEnd DATE = EOMONTH(DATEADD(MONTH, -3, GETDATE()));


    -- Month to insert (last completed month)
    DECLARE @InsertMonthStart DATE = DATEFROMPARTS(
        YEAR(DATEADD(MONTH, -1, GETDATE())),
        MONTH(DATEADD(MONTH, -1, GETDATE())),
        1
    );

    DECLARE @InsertMonthEnd DATE = EOMONTH(DATEADD(MONTH, -1, GETDATE()));


    ---------------------------
    -- 2) DELETE old expired month
    ---------------------------
    DELETE FROM dbo.Orders
    WHERE OrderDate BETWEEN @DeleteMonthStart AND @DeleteMonthEnd;


    ---------------------------
    -- 3) INSERT new month data (with identity)
    ---------------------------
    SET IDENTITY_INSERT dbo.Orders ON;

    INSERT INTO dbo.Orders (OrderID, Product, Amount, OrderDate)
    SELECT 
        OrderID,
        Product,
        Amount,
        OrderDate
    FROM MainDB.dbo.Orders
    WHERE OrderDate BETWEEN @InsertMonthStart AND @InsertMonthEnd;

    SET IDENTITY_INSERT dbo.Orders OFF;

END;
GO

WITH InboundClaims AS (
    SELECT DISTINCT
        ct.ClientTransactionID,
        ct.TransactionID,
        ct.ParentTransactionID,
        ct.PayerIDCode AS InboundPayerID,
        ct.DestinationPartnerCode,
        ct.PayerID,
        MIN(te.EventDate) AS InboundEventDate
    FROM trxQA.dbo.ClientTransactions ct
    JOIN trxQA.dbo.TransactionEvents te 
        ON ct.ClientTransactionID = te.ClientTransactionID
    WHERE te.TransactionDirection = 'Inbound'
        AND ct.SourcePartnerCode IN ('HSD', 'HSI Parents', 'GCC')
        AND ct.SourcePartnerCode <> 'HSI'
        AND te.EventDate BETWEEN '2025-01-01' AND '2025-03-31'
    GROUP BY ct.ClientTransactionID, ct.TransactionID, ct.ParentTransactionID,
             ct.PayerIDCode, ct.DestinationPartnerCode, ct.PayerID
),
OutboundClaims AS (
    SELECT DISTINCT
        ct.ParentTransactionID,
        ct.PayerIDCode AS OutboundPayerID,
        MAX(te.EventDate) AS OutboundEventDate
    FROM trxQA.dbo.ClientTransactions ct
    JOIN trxQA.dbo.TransactionEvents te 
        ON ct.ClientTransactionID = te.ClientTransactionID
    WHERE te.TransactionDirection = 'Outbound'
    GROUP BY ct.ParentTransactionID, ct.PayerIDCode
),
LastStatus AS (
    SELECT 
        ClientTransactionID,
        Description AS LastClaimStatus
    FROM (
        SELECT
            te.ClientTransactionID,
            te.Description,
            ROW_NUMBER() OVER (
                PARTITION BY te.ClientTransactionID
                ORDER BY te.EventDate DESC
            ) AS rn
        FROM trxQA.dbo.TransactionEvents te
    ) sub
    WHERE rn = 1
)
SELECT
    ic.InboundEventDate AS ClaimCreateDate,
    ic.TransactionID AS PartnerClaimID,
    ic.ParentTransactionID AS DCIClaimID,
    ic.InboundPayerID,
    CAST(NULL AS varchar(255)) AS InboundPayerAddress,
    oc.OutboundPayerID,
    ic.DestinationPartnerCode AS OutboundDestinationPartnerCode,
    ls.LastClaimStatus,
    ic.ParentTransactionID AS MetaClaimID,
    ic.PayerID AS PayerIDCode_ID,
    ic.InboundPayerID AS PayerIDCode_Display
FROM InboundClaims ic
LEFT JOIN OutboundClaims oc 
    ON ic.ParentTransactionID = oc.ParentTransactionID
LEFT JOIN LastStatus ls 
    ON ic.ClientTransactionID = ls.ClientTransactionID
WHERE ic.InboundPayerID <> oc.OutboundPayerID  -- Only non-matching
    AND oc.OutboundPayerID IS NOT NULL  -- Has outbound record
ORDER BY 
    ic.InboundEventDate;
